\section{Unconstrained Optimization (5 Points)}

\begin{enumerate}
\item Minimize $f(x) = x^2 - 4x + 4$, determine the critical point, and repor the minimum value of the function. \hfill \textbf{(5 Points)}

\item Optimize $f(x) = x^3 - 3x^2$. Find all the critical points possible. Which one of them are local minima and local maxima? What is the local minimum and maxmium values of $f(x)$? \hfill \textbf{(10 Points)}


\item Consider $f(x, y) = x^2 - y^2$. Compute its gradient $\nabla f$ and associated critical point. Is this critical point a maximum, a minimum, or a saddle point? \hfill \textbf{(5 Points)}

\end{enumerate}

\begin{answerbox}

\begin{enumerate}
\item
\begin{align*}
f(x) & = x^2 - 4x + 4\\
\frac{\partial f(x)}{\partial x} & = 2x - 4 = 0\\
x & = 2\\
\frac{\partial^2 f(x)}{\partial x^2} & = 2 > 0, \quad \text{critical point is a local minimum}
\end{align*}

Minimum value is $f(2) = 2^2 - 4\cdot 2  + 4 = 4 - 8 + 4  = 0$.


\item 

\begin{align*}
\frac{\partial f(x)}{\partial x} & = 3x^2 - 6x  = 0\\
3x(x-2) & = 0 \Rightarrow x= 2 , x = 0\\
\frac{\partial^2 f(x)}{\partial x^2} & = 6x -6 \\
6x - 6 \bigg|_{x = 0} & = -6 <0, \quad \quad \text{critical point is a local maximum} \\
6x - 6 \bigg|_{x = 2} & = 6  > 0 \quad \quad \text{critical point is a local minimum}\\
f(0) & = 0\\
f(2) & = 8 - 3\cdot 4 = -4
\end{align*}

\item 

\begin{align*}
\text{Given } f(x, y) &= x^2 - y^2 \\
\text{The gradient is } \\
\nabla f &= \left( \frac{\partial f}{\partial x}, \frac{\partial f}{\partial y} \right) \\
\frac{\partial f}{\partial x} &= 2x \\
\frac{\partial f}{\partial y} &= -2y \\
\nabla f &= (2x, -2y) \\
\text{Set } \nabla f = (0, 0) \text{ to find critical points:} \\
2x &= 0 \Rightarrow x = 0 \\
-2y &= 0 \Rightarrow y = 0 \\
\text{Critical point: } & (0, 0)
\end{align*}

\begin{align*}
\text{Second derivatives:} \\
\frac{\partial^2 f}{\partial x^2} &= 2 \\
\frac{\partial^2 f}{\partial y^2} &= -2 \\
\frac{\partial^2 f}{\partial x \partial y} &= 0 \\
\text{Hessian matrix: } H &= \begin{bmatrix}
2 & 0 \\
0 & -2
\end{bmatrix} \\
\det(H) &= (2)(-2) - (0)(0) = -4 \\
\end{align*}

$$\text{Since } \det(H) < 0, \text{ the critical point } (0, 0) \text{ is a saddle point.}$$
\end{enumerate}
\end{answerbox}

\section{Constrained Optimization using Lagrange Multiplier (10 Points)}
Using method of Lagrange mulitplier, minimize $f(x, y, z) = x^2 + y^2 + z^2$ subject to $x + 2y + z = 6$.

\begin{enumerate}
\item Set up the Lagrangian.
\item Derive the system of equations.
\item Solve for critical points.
\item Classify extrema.
\item Explain the role of the constraint.
\end{enumerate}

\begin{answerbox}

\subsection*{Step 1: Set up the Lagrangian}
We minimize 
\begin{align*}
f(x, y, z) = x^2 + y^2 + z^2
\end{align*}
subject to the constraint 
\begin{align*}
g(x, y, z) = x + 2y + z = 6.
\end{align*}
The Lagrangian is:
\begin{align*}
\mathcal{L}(x, y, z, \lambda) = x^2 + y^2 + z^2 - \lambda(x + 2y + z - 6)
\end{align*}

\subsection*{Step 2: Derive the system of equations}
\begin{align*}
\frac{\partial \mathcal{L}}{\partial x} &= 2x - \lambda = 0  \\
\frac{\partial \mathcal{L}}{\partial y} &= 2y - 2\lambda = 0 \\
\frac{\partial \mathcal{L}}{\partial z} &= 2z - \lambda = 0  \\
\end{align*}

\subsection*{Step 3: Solve for critical points}
From (1): $x = \frac{\lambda}{2}$ \\
From (2): $y = \lambda$ \\
From (3): $z = \frac{\lambda}{2}$ \\
Substitute into (4):
\begin{align*}
\frac{\lambda}{2} + 2\lambda + \frac{\lambda}{2} = 6 \Rightarrow 3\lambda = 6 \Rightarrow \lambda = 2
\end{align*}
Thus:
\begin{align*}
x = 1, \quad y = 2, \quad z = 1
\end{align*}
Critical point: $(1, 2, 1)$

\subsection*{Step 4: Classify extrema}
The function $f(x, y, z) = x^2 + y^2 + z^2$ is convex (positive definite quadratic form) and the constraint is linear. Therefore, the critical point $(1, 2, 1)$ is a \textbf{minimum}. \\
Minimum value:
\begin{align*}
f(1, 2, 1) = 1^2 + 2^2 + 1^2 = 6
\end{align*}

\subsection*{Step 5: Role of the constraint}
The constraint $x + 2y + z = 6$ defines a plane in $\mathbb{R}^3$. Minimizing $f(x, y, z)$ subject to this constraint finds the point on the plane closest to the origin (since $f$ represents squared distance from origin). The Lagrange multiplier method gives the projection of the origin onto this plane.
\end{answerbox}

\section{Optimizing a Function using Stochastic Gradient Descent (5 Points)}
Use the Stochastic Gradient Descent technique to calculate the optimal value of 
$w_1$ for the cost function shown in Equation (1). Use a learning rate 
$\eta = 0.1$ , and an initial value 
$w_1^{(0)} = 2.0$.
Write down the updated value of 
$w_1$ for 5 iterations using the formula for gradient descent. \newline
This question is \textbf{analytical}, no code is required.
\begin{equation}
    J(w_1) = (w_1-5)^4 + (w_1-3)^2 +  4
\end{equation}

How does it compare with optimizing $J$ using partial differentiation method?
\begin{answerbox}
\noindent The cost function is
$$J(w_1) = (w_1 - 5)^4 + (w_1 - 3)^2 + 4$$
with gradient
$$\frac{\partial J}{\partial w_1} = 4(w_1 - 5)^3 + 2(w_1 - 3).$$
Using stochastic gradient descent (SGD) with learning rate $\eta = 0.1$ and initial value $w_1^{(0)} = 2.0$, the update rule is:
$$w_1^{(t+1)} = w_1^{(t)} - \eta \cdot \frac{\partial J}{\partial w_1}
            = w_1^{(t)} - 0.4\,(w_1^{(t)} - 5)^3 - 0.2\,(w_1^{(t)} - 3).$$

\noindent Iterations:
$$\begin{aligned}
w_1^{(0)} &= 2.0 \\[4pt]
\left.\frac{\partial J}{\partial w_1}\right|_{2.0} &= 4(-3)^3 + 2(-1) = -110
  &&\Rightarrow\; w_1^{(1)} = 2.0 - 0.1(-110) = 13.0 \\[4pt]
\left.\frac{\partial J}{\partial w_1}\right|_{13.0} &= 4(8)^3 + 2(10) = 2068
  &&\Rightarrow\; w_1^{(2)} = 13.0 - 0.1(2068) = -193.8 \\[4pt]
\left.\frac{\partial J}{\partial w_1}\right|_{-193.8} &= 4(-198.8)^3 + 2(-196.8) \approx -3.1428 \times 10^{7}
  &&\Rightarrow\; w_1^{(3)} \approx 3.1426 \times 10^{6} \\[4pt]
\left.\frac{\partial J}{\partial w_1}\right|_{w_1^{(3)}} &\approx 4(3.1426 \times 10^{6})^3 \approx 1.2414 \times 10^{20}
  &&\Rightarrow\; w_1^{(4)} \approx -1.2414 \times 10^{19} \\[4pt]
\left.\frac{\partial J}{\partial w_1}\right|_{w_1^{(4)}} &\approx 4(-1.2414 \times 10^{19})^3 \approx -7.653 \times 10^{57}
  &&\Rightarrow\; w_1^{(5)} \approx 7.653 \times 10^{56}
\end{aligned}$$

\noindent The iterates oscillate in sign and grow without bound, so SGD \textbf{diverges} for $\eta = 0.1$.
The cause is the curvature
$$\frac{\partial^2 J}{\partial w_1^2} = 12(w_1 - 5)^2 + 2,$$
which equals $110$ at the starting point. A gradient step is stable only when $\eta < 2 / J''(w_1)$, which here requires $\eta < 2/110 \approx 0.018$.
The first step overshoots to $w_1 = 13$, where the curvature is larger still ($J'' = 770$), and each later overshoot is worse than the one before.
A smaller learning rate such as $\eta = 0.01$ would converge, with the first step landing at $w_1^{(1)} = 3.1$.

\bigskip
\noindent \textbf{Comparison with Partial Differentiation:} \\
Setting the gradient to zero and substituting $u = w_1 - 5$:
$$4u^3 + 2(u + 2) = 0 \;\implies\; 2u^3 + u + 2 = 0.$$
The cubic has derivative $6u^2 + 1 > 0$, so it has exactly one real root. Cardano's formula gives
$$u = \sqrt[3]{-\tfrac{1}{2} + \sqrt{\tfrac{55}{216}}} + \sqrt[3]{-\tfrac{1}{2} - \sqrt{\tfrac{55}{216}}} \approx -0.8351,$$
so
$$w_1^{*} \approx 4.1649, \qquad J(w_1^{*}) \approx 5.8433.$$
Since $J''(w_1) > 0$ for all $w_1$, $J$ is strictly convex and this stationary point is the unique global minimum.

\noindent Partial differentiation finds the optimum directly, though it requires solving a cubic, which has no simple closed form in practice.
SGD avoids solving the equation, but its success depends on the learning rate.
Convexity guarantees a single minimum, yet it does not guarantee convergence: with $\eta = 0.1$ the method moves away from $w_1^{*} \approx 4.1649$ instead of towards it.
In the quadratic example the curvature was constant, so one fixed $\eta$ worked everywhere. The quartic term here makes the safe step size depend on where the iterates are.
\end{answerbox}

\section{Linear Regression and Regularization (15 points)}
In class we derived and discussed linear regression in detail. Find the result of minimizing the
loss of sum of the squared errors after adding in a penalty called an $L_2$ penalty on the weights.
More formally
\begin{multiequation}
\argmin_\wbf \bigg\{ ||\ybf - \xbf \wbf||^2  + \lambda||\wbf||^2  \bigg\}
\end{multiequation}

which means we are interested in finding $\wbf$ that minimizes the overall expression. How does this change the solution to the original linear regression solution? What is the impact of adding in this penalty?

$|| \cdot ||$ is L2 norm.

\textbf{Hint:} First take the gradient with respect to $\wbf$ and then equate to zero you will obtain an expression for $\wbf$ which is the estimated $\hat{\wbf}$.

Also, remember in vector form the L2 norm $||v||$ can be written as $\vbf^\top \vbf$.


\begin{answerbox}
$$
\argmin_\textbf{w} \bigg\{ ||\textbf{y} - \textbf{x} \textbf{w}||^2  + \lambda||\textbf{w}||^2  \bigg\}
$$

$$
\cfrac{d}{d\textbf{w}}\Bigg\{   \cfrac{1}{2}(\textbf{y} - \textbf{x}\textbf{w})^\top(\textbf{y} - \textbf{x}\textbf{w})  + \cfrac{\lambda}{2}\textbf{w}^\top \textbf{w}\Bigg\} = - \textbf{x}^\top (\textbf{y} -\textbf{x}\textbf{w}) + \lambda \textbf{w} = 0
$$

\begin{align*}
- \textbf{x}^\top (\textbf{y} -\textbf{x}\textbf{w}) + \lambda \textbf{I}\textbf{w} = 0\\
(\textbf{x}^\top\textbf{x} + \lambda\textbf{I})\textbf{w} = \textbf{x}^\top \textbf{y}\\
\textbf{w} = (\textbf{x}^\top\textbf{x} + \lambda \textbf{I})^{-1} \textbf{x}^\top \textbf{y}
\end{align*}

The original solution (without regularization) is $\textbf{w} = (\textbf{x}^\top\textbf{x})^{-1} \textbf{x}^\top \textbf{y}$. Compared to this, with regularization, an extra term is $ \lambda \textbf{I}$. The impact of adding penalty is that it penalizes any data points far from the regressed line, so that the slope of the line is not affected too much, and tends to be less steep. This results in model preventing from overfitting.
\end{answerbox}


\section{Assumptions in Linear Regression (5 Points)}

Write all the assumptions while considering simple linear regression model.

\begin{answerbox}
For the simple linear regression model:
\begin{align*}
Y = \beta_0 + \beta_1 X + \varepsilon
\end{align*}
where $Y$ is the dependent variable, $X$ is the independent variable, $\beta_0$ and $\beta_1$ are parameters, and $\varepsilon$ is the error term.

\begin{enumerate}
    \item \textbf{Linearity}: The relationship between $X$ and $Y$ is linear:
    \begin{align*}
    E(Y|X) = \beta_0 + \beta_1 X
    \end{align*}
    
    \item \textbf{Independence}: Observations $(X_i, Y_i)$ are independently distributed.
    
    \item \textbf{Homoscedasticity}: Constant variance of errors:
    \begin{align*}
    \text{Var}(\varepsilon_i|X_i) = \sigma^2 \quad \text{for all } i
    \end{align*}
    
    \item \textbf{Normality}: The error terms are normally distributed:
    \begin{align*}
    \varepsilon_i \sim N(0, \sigma^2)
    \end{align*}
    
    \item \textbf{No perfect multicollinearity}: In simple linear regression, this means $X$ is not constant (there is variation in $X$ values).
    
    \item \textbf{Exogeneity}: The error term has zero conditional mean:
    \begin{align*}
    E(\varepsilon_i|X_i) = 0
    \end{align*}
    
    \item \textbf{Random sampling}: The sample is randomly drawn from the population.
    
    \item \textbf{No autocorrelation}: Error terms are uncorrelated with each other:
    \begin{align*}
    \text{Cov}(\varepsilon_i, \varepsilon_j) = 0 \quad \text{for } i \neq j
    \end{align*}
    
    \item \textbf{No measurement error}: $X$ is measured without error (or with negligible error).
\end{enumerate}

\end{answerbox}
